    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (raw) {
          const parsed = JSON.parse(raw);
          appState.cashierName = parsed.cashierName || 'Kasir Budi';
          appState.menuItems = Array.isArray(parsed.menuItems) && parsed.menuItems.length
            ? parsed.menuItems
            : JSON.parse(JSON.stringify(DEFAULT_MENU_ITEMS));
          appState.transactions = Array.isArray(parsed.transactions) ? parsed.transactions : [];
          appState.modalAwal = typeof parsed.modalAwal === 'number' ? parsed.modalAwal : 300000;
          appState.shiftReports = Array.isArray(parsed.shiftReports) ? parsed.shiftReports : [];
          return;
        }
      } catch (err) {
        console.warn('Error membaca localStorage:', err);
      }

      appState = createEmptyState();
      saveState();
    }

    // --- CART & CHECKOUT ---
    function getAvailableStockForMenu(menuId) {
      const menu = appState.menuItems.find(m => m.id === menuId);
      if (!menu) return 0;
      if (editingParkedTrxId) {
        const parked = appState.transactions.find(t => t.id === editingParkedTrxId);
        const orig = parked ? parked.items.find(i => i.id === menuId) : null;
        return menu.stock + (orig ? orig.qty : 0);
      }
      return menu.stock;
    }

    function changeCartQty(menuId, delta) {
      const menu = appState.menuItems.find(m => m.id === menuId);
      if (!menu) return;
      const maxAllowed = getAvailableStockForMenu(menuId);
      const idx = cart.findIndex(c => c.id === menuId);

      if (idx === -1 && delta > 0) {
        if (maxAllowed <= 0) {
          showToast(`Stok ${menu.name} habis!`, 'error');
          return;
        }
        cart.push({ id: menu.id, name: menu.name, price: menu.price, qty: 1, note: '' });
      } else if (idx !== -1) {
        const nextQty = cart[idx].qty + delta;
        if (nextQty <= 0) {
          cart.splice(idx, 1);
        } else if (nextQty > maxAllowed) {
          showToast(`Stok ${menu.name} tersisa ${maxAllowed}`, 'warning');
          return;
        } else {
          cart[idx].qty = nextQty;
        }
      }

      renderMenuGrid();
      renderCartPanel();
    }

    function removeCartItem(menuId) {

    }

      }
    }

    function getOrderCalculation() {
      const subtotal = cart.reduce((s, i) => s + i.price * i.qty, 0);
      const totalItems = cart.reduce((s, i) => s + i.qty, 0);
      let discount = 0;
      if (discountAmount > 0 && subtotal > 0) {
        discount = discountType === 'percent'
          ? Math.round((subtotal * Math.min(100, discountAmount)) / 100)
          : Math.min(subtotal, discountAmount);
      }
      const grandTotal = Math.max(0, subtotal - discount);
      return { subtotal, totalItems, discount, grandTotal };
    }

    function syncStockWithOrder(oldItems, newItems) {
      (oldItems || []).forEach(oldIt => {
        const menu = appState.menuItems.find(m => m.id === oldIt.id);
        if (menu) menu.stock += oldIt.qty;
      });
      newItems.forEach(newIt => {
        const menu = appState.menuItems.find(m => m.id === newIt.id);
        if (menu) menu.stock = Math.max(0, menu.stock - newIt.qty);
      });
    }

    function parkTableOrder() {
      if (cart.length === 0) return;
      const { subtotal, discount, grandTotal } = getOrderCalculation();
      const tableLabel = selectedTable || 'Meja Hidang';
      const custLabel = customerName.trim() || `${tableLabel} (Hidang)`;

      if (editingParkedTrxId) {
        const existing = appState.transactions.find(t => t.id === editingParkedTrxId);
        if (existing) {
          syncStockWithOrder(existing.items, cart);
          existing.mode = 'Makan di Tempat';
          existing.customerName = custLabel;
          existing.table = tableLabel;
          existing.items = cart.map(c => ({ ...c }));
          existing.note = globalNote;
          existing.subtotal = subtotal;
          existing.discount = discount;
          existing.total = grandTotal;
          existing.time = getCurrentTime();
        }
        showToast(`Pesanan ${custLabel} diperbarui.`);
      } else {
        syncStockWithOrder([], cart);
        appState.transactions.unshift({
          id: getNextReceiptNumber(),
          time: getCurrentTime(),
          mode: 'Makan di Tempat',
          customerName: custLabel,
          table: tableLabel,
          items: cart.map(c => ({ ...c })),
          note: globalNote,
          subtotal,
          discount,
          total: grandTotal,
          method: 'Belum Bayar',
          cashReceived: 0,
          change: 0,
          status: 'Hidang Meja',
          voidReason: ''
        });
        showToast(`${custLabel} disimpan ke Hidang Meja!`);
      }

      saveState();
      resetCurrentOrder();
    }

    function processPayment() {
      if (cart.length === 0) return;
      const { subtotal, discount, grandTotal } = getOrderCalculation();

      if (paymentMethod === 'Tunai' && cashReceived < grandTotal) {
        showToast('Uang tunai kurang dari total pembayaran!', 'error');
        return;
      }

      const finalCash = paymentMethod === 'Tunai' ? cashReceived : grandTotal;
      const finalChange = paymentMethod === 'Tunai' ? Math.max(0, cashReceived - grandTotal) : 0;
      const custLabel = customerName.trim() || selectedTable || (orderMode === 'Bungkus' ? 'Pelanggan Bungkus' : 'Tamu Warung');

      let completedTrx = null;

      if (editingParkedTrxId) {
        const existing = appState.transactions.find(t => t.id === editingParkedTrxId);
        if (existing) {
          syncStockWithOrder(existing.items, cart);
          existing.time = getCurrentTime();
          existing.mode = orderMode;
          existing.customerName = custLabel;
          existing.table = selectedTable;
          existing.items = cart.map(c => ({ ...c }));
          existing.note = globalNote;
          existing.subtotal = subtotal;
          existing.discount = discount;
          existing.total = grandTotal;
          existing.method = paymentMethod;
          existing.cashReceived = finalCash;
          existing.change = finalChange;
          existing.status = 'Lunas';
          completedTrx = existing;
        }
      } else {
        syncStockWithOrder([], cart);
        completedTrx = {
          id: getNextReceiptNumber(),
          time: getCurrentTime(),
          mode: orderMode,
          customerName: custLabel,
          table: selectedTable,
          items: cart.map(c => ({ ...c })),
          note: globalNote,
          subtotal,
          discount,
          total: grandTotal,
          method: paymentMethod,
          cashReceived: finalCash,
          change: finalChange,
          status: 'Lunas',
          voidReason: ''
        };
        appState.transactions.unshift(completedTrx);
      }

      saveState();
      resetCurrentOrder();
      viewReceiptModal(completedTrx.id);
    }

      const reason = document.getElementById('void-reason-input').value.trim();
      if (!reason) {
        showToast('Alasan pembatalan wajib diisi!', 'error');
        return;

      const trx = appState.transactions.find(t => t.id === voidTargetId);
      if (trx && trx.status !== 'Batal') {

                class="w-14 py-1 px-1.5 text-center border border-zinc-300 rounded-xl text-xs font-extrabold tabular-nums ${
                  item.stock === 0 ? 'bg-red-50 text-red-700 border-red-300' : item.stock <= 5 ? 'bg-amber-50 text-amber-800 border-amber-300' : 'bg-white'
                }"
              />
              <button onclick="quickRestock('${item.id}', 5)" class="px-2.5 h-7 rounded-full bg-[#D5D2EE] hover:bg-[#18181B] text-[#18181B] hover:text-white text-[11px] font-bold">+5</button>
            </div>
          </td>
          <td class="py-3 px-4 text-right">
            <div class="inline-flex items-center gap-1.5">
              <button onclick="toggleMenuStatus('${item.id}')" class="px-2.5 py-1 rounded-full text-[10px] font-extrabold ${
                item.active ? 'bg-[#B8E6D0] text-[#065F46]' : 'bg-zinc-200 text-zinc-600'
              }">
                ${item.active ? 'Aktif' : 'Nonaktif'}
              </button>
              <button onclick="openMenuModal('${item.id}')" class="px-2.5 py-1 rounded-full bg-zinc-100 hover:bg-zinc-200 text-[#18181B] text-[11px] font-bold">Edit</button>
            </div>
          </td>
        </tr>
      `).join('');
    }

    function quickRestock(id, delta) {
      const menu = appState.menuItems.find(m => m.id === id);
      if (!menu) return;
      menu.stock = Math.max(0, menu.stock + delta);
      saveState();
      renderAll();
      if (delta > 0) showToast(`Stok ${menu.name} ditambah +${delta}`);
    }

    function setStockDirectly(id, val) {
      const menu = appState.menuItems.find(m => m.id === id);
      if (!menu) return;
      menu.stock = Math.max(0, parseInt(val, 10) || 0);
      saveState();
      renderAll();
    }

    function toggleMenuStatus(id) {
      const menu = appState.menuItems.find(m => m.id === id);
      if (!menu) return;
      menu.active = !menu.active;
      saveState();
      renderAll();
      showToast(`${menu.name} ${menu.active ? 'diaktifkan' : 'dinonaktifkan'}.`);
    }

        const m = appState.menuItems.find(x => x.id === id);
        heading.textContent = 'Edit Hidangan';
        document.getElementById('menu-edit-id').value = m.id;
        document.getElementById('menu-name-field').value = m.name;

        document.getElementById('menu-price-field').value = m.price;
        document.getElementById('menu-stock-field').value = m.stock;

        document.getElementById('menu-name-field').value = '';
        document.getElementById('menu-cat-field').value = 'Daging & Ayam';

    function handleSaveMenu(e) {
      e.preventDefault();
      const id = document.getElementById('menu-edit-id').value;
      const name = document.getElementById('menu-name-field').value.trim();
      const category = document.getElementById('menu-cat-field').value;
      const price = Math.max(500, parseInt(document.getElementById('menu-price-field').value, 10) || 0);
      const stock = Math.max(0, parseInt(document.getElementById('menu-stock-field').value, 10) || 0);
      const plate = document.getElementById('menu-plate-field').value;
      const active = document.getElementById('menu-active-field').checked;

      if (id) {
        const m = appState.menuItems.find(x => x.id === id);
        if (m) Object.assign(m, { name, category, price, stock, plate, active });
        showToast(`Menu "${name}" diperbarui.`);
      } else {
        appState.menuItems.unshift({
          id: 'm' + Date.now(),
          name,
          category,
          price,
          stock,
          plate,
          active
        });
        showToast(`Menu "${name}" ditambahkan.`);
      }

      saveState();
      closeMenuModal();
      renderAll();
    }

    // --- TAB 4: SHIFT RECAP & EXPORT/IMPORT BACKUP ---
    function getShiftMetrics() {
      const lunas = appState.transactions.filter(t => t.status === 'Lunas');
      const batal = appState.transactions.filter(t => t.status === 'Batal');
      const hidang = appState.transactions.filter(t => t.status === 'Hidang Meja');

      const netRevenue = lunas.reduce((s, t) => s + t.total, 0);
      const tunaiTotal = lunas.filter(t => t.method === 'Tunai').reduce((s, t) => s + t.total, 0);
      const qrisTotal = lunas.filter(t => t.method === 'QRIS').reduce((s, t) => s + t.total, 0);
      const transferTotal = lunas.filter(t => t.method === 'Transfer').reduce((s, t) => s + t.total, 0);

      const dishMap = {};
      lunas.forEach(t => {
        t.items.forEach(it => {
          if (!dishMap[it.name]) dishMap[it.name] = { name: it.name, qty: 0, total: 0 };
          dishMap[it.name].qty += it.qty;
          dishMap[it.name].total += it.price * it.qty;
        });
      });

      const topDishes = Object.values(dishMap).sort((a, b) => b.qty - a.qty).slice(0, 5);
      return { lunasCount: lunas.length, batalCount: batal.length, hidangCount: hidang.length, netRevenue, tunaiTotal, qrisTotal, transferTotal, topDishes };
    }

    function evaluateShiftVariance() {
      const raw = document.getElementById('shift-physical-input').value;
      const card = document.getElementById('shift-variance-card');
      const amt = document.getElementById('shift-variance-amount');
      const desc = document.getElementById('shift-variance-desc');

      if (raw === '') {
        card.className = 'p-3.5 rounded-xl bg-white/90 border border-white';
        amt.className = 'text-lg font-extrabold text-[#18181B] tabular-nums mt-0.5';
        amt.textContent = 'Belum dihitung';
        desc.textContent = 'Masukkan nominal uang tunai fisik.';
        return;
      }

      const m = getShiftMetrics();
      const expected = appState.modalAwal + m.tunaiTotal;
      const physical = Math.max(0, Number(raw) || 0);
      const diff = physical - expected;

      if (diff === 0) {
        card.className = 'p-3.5 rounded-xl bg-[#B8E6D0] border border-emerald-300';
        amt.className = 'text-lg font-extrabold text-[#065F46] tabular-nums mt-0.5';
        amt.textContent = 'Rp 0 (PAS)';
        desc.textContent = 'Kas tunai cocok 100%.';
      } else if (diff > 0) {
        card.className = 'p-3.5 rounded-xl bg-[#D9F27E] border border-lime-300';
        amt.className = 'text-lg font-extrabold text-[#3F6212] tabular-nums mt-0.5';
        amt.textContent = `+ ${formatRupiah(diff)} (LEBIH)`;
        desc.textContent = 'Uang fisik lebih dari catatan sistem.';
      } else {
        card.className = 'p-3.5 rounded-xl bg-[#FCA5A5] border border-red-300';
        amt.className = 'text-lg font-extrabold text-[#7F1D1D] tabular-nums mt-0.5';
        amt.textContent = `- ${formatRupiah(Math.abs(diff))} (KURANG)`;
        desc.textContent = 'Selisih kurang! Periksa kembali kas laci.';
      }
    }

    function saveShiftClosing() {
      const raw = document.getElementById('shift-physical-input').value;
      if (raw === '') {
        showToast('Isi uang tunai fisik terlebih dahulu!', 'error');
        return;
      }

      const m = getShiftMetrics();
      const expected = appState.modalAwal + m.tunaiTotal;
      const physical = Math.max(0, Number(raw) || 0);
      const diff = physical - expected;
      const note = document.getElementById('shift-note-input').value.trim();

      appState.shiftReports.unshift({
        date: new Date().toLocaleDateString('id-ID'),
        time: getCurrentTime(),
        cashier: appState.cashierName,
        expected,
        physical,
        diff,
        netRevenue: m.netRevenue,
        note
      });

      saveState();
      renderSavedShiftReports();
      showToast('Laporan tutup shift tersimpan!');
    }

    function renderSavedShiftReports() {
      const box = document.getElementById('saved-shift-reports');
      if (!appState.shiftReports.length) {
        box.classList.add('hidden');
        return;
      }
      box.classList.remove('hidden');
      box.innerHTML = `
        <h4 class="text-xs font-extrabold text-[#18181B] mb-2">Riwayat Tutup Shift:</h4>
        <div class="space-y-2">
          ${appState.shiftReports.map(r => `
            <div class="bg-white/90 rounded-xl p-3 text-xs flex flex-col sm:flex-row sm:items-center justify-between gap-2">
              <div>
                <span class="font-extrabold text-[#18181B]">${r.date || ''} ${r.time} • ${r.cashier}</span>
                <span class="text-zinc-500 ml-1.5">Omzet: ${formatRupiah(r.netRevenue)}</span>
                ${r.note ? `<div class="text-zinc-500 mt-0.5">Catatan: ${r.note}</div>` : ''}
              </div>
              <div class="font-extrabold tabular-nums ${r.diff === 0 ? 'text-[#065F46]' : r.diff > 0 ? 'text-[#3F6212]' : 'text-[#7F1D1D]'}">
                Fisik: ${formatRupiah(r.physical)} (${r.diff === 0 ? 'PAS' : (r.diff > 0 ? '+' : '-') + formatRupiah(Math.abs(r.diff))})
              </div>
            </div>
          `).join('')}
        </div>
      `;
    }

    // --- BACKUP & EXPORT (JSON + CSV) ---
    function downloadFile(content, filename, mimeType) {
      const blob = new Blob([content], { type: mimeType });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      setTimeout(() => URL.revokeObjectURL(url), 1000);
    }

    function exportBackupJSON() {
      const dateStamp = new Date().toISOString().slice(0, 10);
      downloadFile(JSON.stringify(appState, null, 2), `backup-salero-minang-${dateStamp}.json`, 'application/json');
      showToast('Backup JSON berhasil diunduh!');
    }

    function importBackupJSON(event) {
      const file = event.target.files && event.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = (e) => {
        try {
          const parsed = JSON.parse(e.target.result);
          if (!parsed || !Array.isArray(parsed.menuItems) || !Array.isArray(parsed.transactions)) {
            showToast('Format file backup tidak valid!', 'error');
            return;
          }
          appState = {
            cashierName: parsed.cashierName || 'Kasir Budi',
            menuItems: parsed.menuItems,
            transactions: parsed.transactions,
            modalAwal: typeof parsed.modalAwal === 'number' ? parsed.modalAwal : 300000,
            shiftReports: Array.isArray(parsed.shiftReports) ? parsed.shiftReports : []
          };
          saveState();
          resetCurrentOrder();
          renderAll();
          showToast('Data berhasil dipulihkan dari backup JSON!');
        } catch (err) {
          showToast('Gagal membaca file JSON!', 'error');
        }
      };
      reader.readAsText(file);
      event.target.value = '';
    }

    function exportTransactionsCSV() {
      const headers = ['No Struk', 'Jam', 'Mode', 'Pelanggan/Meja', 'Metode', 'Subtotal', 'Diskon', 'Total', 'Status', 'Item'];
      const rows = appState.transactions.map(t => {
        const itemsStr = t.items.map(i => `${i.qty}x ${i.name}`).join('; ');
        return [
          t.id,
          t.time,
          t.mode,
          t.customerName || t.table || '-',
          t.method,
          t.subtotal,
          t.discount,
          t.total,
          t.status,
          itemsStr
        ].map(cell => `"${String(cell ?? '').replace(/"/g, '""')}"`).join(',');
      });
      const csv = [headers.join(','), ...rows].join('\n');
      const dateStamp = new Date().toISOString().slice(0, 10);
      downloadFile(csv, `transaksi-salero-minang-${dateStamp}.csv`, 'text/csv;charset=utf-8;');
      showToast('CSV transaksi berhasil diunduh!');
    }
